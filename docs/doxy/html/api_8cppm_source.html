403template <class T>
404concept EnvParsable = std::same_as<T, bool> || std::integral<T> || std::floating_point<T> ||
405 std::same_as<T, std::string>;
406
416template <EnvParsable T>
417T env(std::string_view name, T def) {
418 const auto value = env(name);
419 if (!value || value->empty())
420 return def;
421 if constexpr (std::same_as<T, bool>) {
422 std::string lower;
423 lower.reserve(value->size());
424 for (const char c : *value)
425 lower.push_back(c >= 'A' && c <= 'Z' ? static_cast<char>(c - 'A' + 'a') : c);
426 if (lower == "1" || lower == "true" || lower == "on" || lower == "yes" || lower == "y")
427 return true;
428 if (lower == "0" || lower == "false" || lower == "off" || lower == "no" || lower == "n")
429 return false;
430 return def;
431 } else if constexpr (std::same_as<T, std::string>) {
432 return *value;
433 } else {
434 T parsed {};
435 const auto [end, ec] = std::from_chars(value->data(), value->data() + value->size(), parsed);
436 if (ec == std::errc {} && end == value->data() + value->size())
437 return parsed;
438 return def;
439 }
440}
